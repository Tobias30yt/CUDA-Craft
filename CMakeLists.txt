cmake_minimum_required(VERSION 3.25)
project(CUDACraft VERSION 0.11.5 LANGUAGES C CXX)

# The portable Windows pack must not depend on an installed VC++ redistributable.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CUDACRAFT_CUDA AUTO CACHE STRING "CUDA backend: AUTO, ON or OFF")
set_property(CACHE CUDACRAFT_CUDA PROPERTY STRINGS AUTO ON OFF)
if(NOT CUDACRAFT_CUDA MATCHES "^(AUTO|ON|OFF)$")
  message(FATAL_ERROR "CUDACRAFT_CUDA must be AUTO, ON or OFF")
endif()

add_library(cudacraft_protocol src/protocol.cpp)
target_include_directories(cudacraft_protocol PUBLIC include)
set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZLIB_INSTALL OFF CACHE BOOL "" FORCE)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/.tools/dependencies/zlib-1.4/CMakeLists.txt")
  add_subdirectory(.tools/dependencies/zlib-1.4 _deps/zlib)
else()
  include(FetchContent)
  FetchContent_Declare(zlib URL https://zlib.net/zlib14.zip
    URL_HASH SHA256=38bc7ed1c8ac3c870a159623deacf74a46c393ae1ad23f1dca1ed8e272deb058
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(zlib)
endif()
add_library(cudacraft_compression src/compression.cpp)
target_link_libraries(cudacraft_compression PUBLIC cudacraft_protocol PRIVATE zlibstatic)
add_library(cudacraft_net src/server.cpp)
target_link_libraries(cudacraft_net PUBLIC cudacraft_protocol)
if(WIN32)
  set(hc_root third_party/hyperchunk/core)
  add_library(cudacraft_density STATIC
    ${hc_root}/src/arena.c ${hc_root}/src/chunk.c ${hc_root}/src/rng.c ${hc_root}/src/md5.c ${hc_root}/src/json.c
    ${hc_root}/src/noise_perlin.c ${hc_root}/src/noise_octaves.c ${hc_root}/src/noise_blended.c ${hc_root}/src/noise_normal.c
    ${hc_root}/src/df_compile.c ${hc_root}/src/df_eval.c ${hc_root}/src/df_isa.c
    ${hc_root}/src/df_simd_avx2.c ${hc_root}/src/df_simd_avx512.c
    ${hc_root}/src/noise_chunk.c ${hc_root}/src/aquifer.c ${hc_root}/src/ore_veins.c ${hc_root}/src/gen_noise_stage.c
    ${hc_root}/src/beard.c ${hc_root}/src/blocks.c ${hc_root}/src/counters.c
    ${hc_root}/src/surface.c ${hc_root}/src/surface_compile.c ${hc_root}/src/biome_zoom.c ${hc_root}/src/biome_temp.c
    ${hc_root}/src/carvers.c ${hc_root}/src/gen_carvers_stage.c
    ${hc_root}/src/sha256.c ${hc_root}/src/sha256_ni.c)
  target_include_directories(cudacraft_density SYSTEM PUBLIC third_party/hyperchunk ${hc_root}/include)
  set_target_properties(cudacraft_density PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
  if(MSVC)
    target_compile_options(cudacraft_density PRIVATE /fp:precise /experimental:c11atomics /wd4244 /wd4267 /utf-8)
  else()
    target_compile_options(cudacraft_density PRIVATE -fwrapv -ffp-contract=off)
    target_link_libraries(cudacraft_density PUBLIC m)
  endif()
  add_library(cubiomes STATIC third_party/cubiomes/generator.c third_party/cubiomes/layers.c
    third_party/cubiomes/biomenoise.c third_party/cubiomes/biomes.c third_party/cubiomes/noise.c)
  target_include_directories(cubiomes SYSTEM PUBLIC third_party/cubiomes)
  set_target_properties(cubiomes PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
  if(MSVC)
    # /fp:precise preserves operation order; no /fp:contract or /arch:AVX is enabled.
    # /fp:strict rejects the upstream C static floating-point division constants.
    target_compile_options(cubiomes PRIVATE /fp:precise /wd4244 /wd4267)
  else()
    target_compile_options(cubiomes PRIVATE -fwrapv -ffp-contract=off)
    target_link_libraries(cubiomes PUBLIC m)
  endif()
  target_link_libraries(cudacraft_net PRIVATE ws2_32)
  add_library(cudacraft_login src/crypto_windows.cpp src/auth.cpp src/auth_windows.cpp src/login.cpp)
  target_link_libraries(cudacraft_login PUBLIC cudacraft_compression PRIVATE bcrypt winhttp)
  add_library(cudacraft_game src/world.cpp src/dimensions.cpp src/block_mechanics.cpp src/game.cpp src/gameplay.cpp src/crafting.cpp src/inventory_actions.cpp src/containers.cpp src/simulation.cpp src/admin.cpp src/performance.cpp src/terrain.cpp src/biome_source.cpp src/climate_terrain.cpp src/noise_terrain.cpp)
  target_link_libraries(cudacraft_game PUBLIC cudacraft_protocol cudacraft_gpu PRIVATE cubiomes cudacraft_density)
  target_sources(cudacraft_game PRIVATE src/world_lock.cpp)
  target_include_directories(cudacraft_game SYSTEM PRIVATE third_party)
  target_link_libraries(cudacraft_game PRIVATE psapi)
  target_sources(cudacraft_net PRIVATE src/console.cpp)
  target_link_libraries(cudacraft_login PUBLIC cudacraft_game)
  target_include_directories(cudacraft_login SYSTEM PRIVATE third_party)
  target_link_libraries(cudacraft_net PUBLIC cudacraft_login)
  target_compile_definitions(cudacraft_net PUBLIC CUDACRAFT_LOGIN=1)
endif()

if(NOT CUDACRAFT_CUDA STREQUAL "OFF")
  include(CheckLanguage)
  check_language(CUDA)
  if(CMAKE_CUDA_COMPILER)
    enable_language(CUDA)
    find_package(CUDAToolkit REQUIRED)
  elseif(CUDACRAFT_CUDA STREQUAL "ON")
    message(FATAL_ERROR "CUDA requested, but nvcc was not found. Install the NVIDIA CUDA Toolkit or set CMAKE_CUDA_COMPILER.")
  endif()
endif()
if(CMAKE_CUDA_COMPILER AND NOT CUDACRAFT_CUDA STREQUAL "OFF")
  add_library(cudacraft_gpu src/gpu/backend.cu)
  target_link_libraries(cudacraft_gpu PRIVATE CUDA::cudart_static)
  set_target_properties(cudacraft_gpu PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED ON)
  message(STATUS "CUDA Craft: CUDA diagnostics enabled")
else()
  add_library(cudacraft_gpu src/gpu/unavailable.cpp)
  message(STATUS "CUDA Craft: networking-only build; GPU simulation is not implemented")
endif()
target_include_directories(cudacraft_gpu PUBLIC include)

add_executable(cuda-craft src/main.cpp)
target_link_libraries(cuda-craft PRIVATE cudacraft_net cudacraft_gpu)
set(cudacraft_targets cudacraft_protocol cudacraft_compression cudacraft_net cuda-craft)
if(WIN32)
  list(APPEND cudacraft_targets cudacraft_login cudacraft_game)
endif()
foreach(target ${cudacraft_targets})
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
